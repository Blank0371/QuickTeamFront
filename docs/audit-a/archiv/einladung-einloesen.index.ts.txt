import { createClient } from "jsr:@supabase/supabase-js@2";

Deno.serve(async (req) => {
  try {
    const { hash } = await req.json();
    if (!hash) return json({ error: "hash erforderlich" }, 400);

    const admin = createClient(
      Deno.env.get("SUPABASE_URL")!,
      Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
    );

    // --- validate invite ---
    const { data: einladung } = await admin
      .from("einladungen")
      .select("hash, betrieb_id, mitarbeiter_id, ablaufdatum, eingeloest_am")
      .eq("hash", hash)
      .maybeSingle();

    if (!einladung) return json({ error: "Einladung ungültig" }, 404);
    if (einladung.eingeloest_am) return json({ error: "Einladung bereits eingelöst" }, 409);
    if (new Date(einladung.ablaufdatum) < new Date()) {
      return json({ error: "Einladung abgelaufen" }, 410);
    }

    // --- find or create the auth user for this Mitarbeiter ---
    const { data: ma } = await admin
      .from("mitarbeiter")
      .select("id, auth_id")
      .eq("id", einladung.mitarbeiter_id)
      .single();

    const email = `mitarbeiter-${einladung.mitarbeiter_id}@invite.local`;
    let authId = ma?.auth_id as string | null;

    if (!authId) {
      const { data: created, error: cErr } = await admin.auth.admin.createUser({
        email,
        email_confirm: true,
        user_metadata: {
          rolle: "mitarbeiter",
          mitarbeiter_id: einladung.mitarbeiter_id,
          betrieb_id: einladung.betrieb_id,
        },
      });
      if (cErr || !created.user) return json({ error: cErr?.message ?? "User-Anlage fehlgeschlagen" }, 500);
      authId = created.user.id;

      await admin.from("mitarbeiter")
        .update({ auth_id: authId })
        .eq("id", einladung.mitarbeiter_id);
    }

    // --- mint a session (magiclink → verifyOtp) ---
    const { data: link, error: lErr } = await admin.auth.admin.generateLink({
      type: "magiclink",
      email,
    });
    if (lErr || !link.properties?.hashed_token) {
      return json({ error: lErr?.message ?? "Token-Erzeugung fehlgeschlagen" }, 500);
    }

    const anon = createClient(
      Deno.env.get("SUPABASE_URL")!,
      Deno.env.get("SUPABASE_ANON_KEY")!
    );
    const { data: session, error: vErr } = await anon.auth.verifyOtp({
      token_hash: link.properties.hashed_token,
      type: "email",
    });
    if (vErr || !session.session) {
      return json({ error: vErr?.message ?? "Session-Erzeugung fehlgeschlagen" }, 500);
    }

    // --- mark redeemed (only after the session exists) ---
    const { error: updErr } = await admin
      .from("einladungen")
      .update({ eingeloest_am: new Date().toISOString() })
      .eq("hash", hash)
      .is("eingeloest_am", null);   // guards against a double-redeem race
    if (updErr) return json({ error: updErr.message }, 500);

    return json({
      access_token: session.session.access_token,
      refresh_token: session.session.refresh_token,
      mitarbeiter_id: einladung.mitarbeiter_id,
      betrieb_id: einladung.betrieb_id,
    });
  } catch (e) {
    return json({ error: String(e) }, 500);
  }
});

function json(body: unknown, status = 200) {
  return new Response(JSON.stringify(body), {
    status,
    headers: { "Content-Type": "application/json" },
  });
}
